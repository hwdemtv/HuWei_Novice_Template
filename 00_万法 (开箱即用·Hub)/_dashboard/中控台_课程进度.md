# 🎓 课程进度追踪

```dataviewjs
const content = await dv.io.load("课程进度追踪.md");
const lines = content.split('\n');

let tier0 = {total:0, done:0}, tier1 = {total:0, done:0};
let tier2 = {total:0, done:0}, tier3 = {total:0, done:0};

for (const line of lines) {
    const trimmed = line.trim();
    if (!trimmed.startsWith('| L')) continue;
    const cols = trimmed.split('|').map(c => c.trim()).filter(c => c);
    if (cols.length < 3) continue;
    
    const code = cols[0];
    const status = cols[2] || '';
    const isDone = status.includes('✅');
    
    if (code.startsWith('L0')) { tier0.total++; if(isDone) tier0.done++; }
    else if (code.startsWith('L1')) { tier1.total++; if(isDone) tier1.done++; }
    else if (code.startsWith('L2')) { tier2.total++; if(isDone) tier2.done++; }
    else if (code.startsWith('L3')) { tier3.total++; if(isDone) tier3.done++; }
}

const totalDone = tier0.done + tier1.done + tier2.done + tier3.done;
const totalAll = tier0.total + tier1.total + tier2.total + tier3.total;
const pct = totalAll > 0 ? Math.round(totalDone / totalAll * 100) : 0;

function bar(done, total) {
    if (total === 0) return '—';
    const p = Math.round(done / total * 100);
    const fill = '█'.repeat(Math.min(Math.floor(p / 10), 10));
    const empty = '░'.repeat(Math.max(10 - Math.floor(p / 10), 0));
    return `${fill}${empty} ${p}%`;
}

dv.paragraph(`## 总进度 **${pct}%** — ${totalDone}/${totalAll} 课已完成`);

dv.table(
  ["阶段", "名称", "进度", "完成率"],
  [
    ["🧱 Tier 0", "筑基", `${tier0.done}/${tier0.total}`, bar(tier0.done, tier0.total)],
    ["🔥 Tier 1", "点火", `${tier1.done}/${tier1.total}`, bar(tier1.done, tier1.total)],
    ["⚗️ Tier 2", "炼器", `${tier2.done}/${tier2.total}`, bar(tier2.done, tier2.total)],
    ["🏔️ Tier 3", "归藏", `${tier3.done}/${tier3.total}`, bar(tier3.done, tier3.total)],
  ]
);
```
